Considere que la Base de Datos es para una aplicación CRUD que además permita consultar literatura formal 
afín a las Ciencias e Ingeniería en Computación y generar los respectivos formatos especialmente APA 7, IEEE, 
ACM y adicional Chicago, Harvard, etc.


// --- SISTEMA DE GESTIÓN BIBLIOGRÁFICA (EXCEL COMPLIANCE) ---

Table roles {
  id int [pk, increment]
  nombre varchar(50) [not null]
}

Table usuarios {
  id int [pk, increment]
  rol_id int [ref: > roles.id]
  nombre varchar(100)
  correo varchar(255) [unique]
  contrasena varchar(255)
  created_at timestamp
}

// --- CATALOGOS ACADÉMICOS Y GEOGRÁFICOS ---

Table areas {
  id int [pk, increment]
  nombre varchar(100) [not null, note: 'Ej. Ingeniería, Ciencias Sociales']
}

Table materias {
  id int [pk, increment]
  area_id int [ref: > areas.id]
  nombre varchar(100) [not null, note: 'Ej. Bases de Datos, Termodinámica']
}

Table paises {
  id int [pk, increment]
  nombre varchar(100) [not null]
  codigo_iso varchar(5)
}

Table tipos_referencia {
  id int [pk, increment]
  nombre varchar(50) [not null, note: 'Ej. Libro, Artículo, Tesis']
}

// --- ENTIDAD PRINCIPAL (REFERENCIAS) ---

Table referencias {
  id int [pk, increment]
  tipo_referencia_id int [ref: > tipos_referencia.id]
  materia_id int [ref: > materias.id]
  pais_id int [ref: > paises.id]
  titulo varchar(255) [not null]
  editorial varchar(150)
  institucion varchar(255) [note: 'Proveniente de Excel']
  tema varchar(255) [note: 'Palabras clave o sub-tema específico']
  fecha_publicacion date
  isbn_issn varchar(20) [unique]
  doi varchar(100) [unique]
  url varchar(500)
  resumen text
  created_at timestamp
}

// --- AUTORES ---

Table autores {
  id int [pk, increment]
  nombre varchar(100)
  apellidos varchar(150)
  nota_biografica text
}

// Relación N:M entre Autores y Referencias
Table referencia_autor {
  referencia_id int [ref: > referencias.id]
  autor_id int [ref: > autores.id]
  orden tinyint [note: '1: Principal, 2: Co-autor...']

  indexes {
    (referencia_id, autor_id) [pk]
  }
}

// --- GESTIÓN DE ARCHIVOS ---

Table archivos {
  id int [pk, increment]
  referencia_id int [ref: > referencias.id]
  nombre_archivo varchar(255)
  ruta_storage varchar(500)
  formato varchar(10) [default: 'pdf']
  tamano_bytes bigint
}

// --- BIBLIOTECA PERSONAL (COLECCIONES) ---

Table colecciones {
  id int [pk, increment]
  usuario_id int [ref: > usuarios.id]
  referencia_id int [ref: > referencias.id]
  fecha_agregado timestamp
  comentario_personal text
}
